cmake_minimum_required(VERSION 3.28)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

find_program(CLANG_FORMAT_EXE NAMES clang-format REQUIRED)

project(dnsd LANGUAGES C)

set(HEADER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/include")
set(SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src")
set(CLANG_FORMAT_CFG "${CMAKE_CURRENT_SOURCE_DIR}/.clang-format")

file(GLOB_RECURSE HEADERS CONFIGURE_DEPENDS "${HEADER_DIR}/*.h")
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${SOURCE_DIR}/*.c")

add_executable(${PROJECT_NAME} ${SOURCES})
target_include_directories(${PROJECT_NAME} PRIVATE ${HEADER_DIR})

add_custom_target(clang-format
  COMMAND ${CLANG_FORMAT_EXE} -i -style=file:${CLANG_FORMAT_CFG} ${SOURCES} ${HEADERS}
)
